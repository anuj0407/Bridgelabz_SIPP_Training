CLASS Doctor
    name
    specialization
    experienceYears
    
    METHOD Constructor(name, specialization, experienceYears)
        SET fields accordingly
    
    METHOD getDetails()
        RETURN name + " - " + specialization + " (" + experienceYears + " years)"

DECLARE departmentDirectory as Map<String, List<Doctor>>

// ----- Simulation -----
departmentDirectory["Cardiology"] = [Doctor("Dr. Mehta", "Cardiologist", 15),
                                     Doctor("Dr. Kapoor", "Cardiologist", 10)]
departmentDirectory["Neurology"] = [Doctor("Dr. Sharma", "Neurologist", 10),
                                    Doctor("Dr. Das", "Neurologist", 8)]
departmentDirectory["Pediatrics"] = [Doctor("Dr. Ritu", "Pediatrician", 12),
                                     Doctor("Dr. Anjali", "Pediatrician", 7)]

// Patient asks for Neurology doctors
doctors = departmentDirectory["Neurology"]
FOR each doctor IN doctors
    PRINT doctor.getDetails()
